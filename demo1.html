import time
import board
import adafruit_dht

dhtDevice = adafruit_dht.DHT11(board.D4)

print("开始读取 DHT11 温湿度数据，按 Ctrl+C 停止")

while True:
    try:
        temperature = dhtDevice.temperature  # 温度，单位 °C
        humidity = dhtDevice.humidity        # 湿度，单位 %
        print(f"温度: {temperature}°C  湿度: {humidity}%")
    except RuntimeError as error:
        # DHT11 读取偶尔会失败，捕获异常并重试
        print("读取失败，重试中...")
    time.sleep(2)  # 每2秒读取一次

开发MQTT客户端并上传数据 ​

ip install paho-mqtt adafruit-circuitpython-dht
sudo apt install python3-rpi.gpio -y
python
import time
import json
import board
import adafruit_dht
import paho.mqtt.client as mqtt

# ---------- DHT11 初始化 ----------
dhtDevice = adafruit_dht.DHT11(board.D4)

# ---------- MQTT 配置 ----------
MQTT_BROKER = "192.168.1.20"
MQTT_PORT = 1883
CLIENT_ID = "device_1958463764077232128"
USERNAME = "AIotFlow"
PASSWORD = "1958463764077232129"
SEND_TOPIC = "/1/1958463764077232128/sub"   # 用于发送数据
RECV_TOPIC = "/1/1958463764077232128/pub"   # 用于接收消息

# ---------- MQTT 客户端 ----------
client = mqtt.Client(client_id=CLIENT_ID, clean_session=True, protocol=mqtt.MQTTv311)
client.username_pw_set(USERNAME, PASSWORD)
client.reconnect_delay_set(min_delay=1, max_delay=120)

# ---------- 回调函数 ----------
def on_connect(client, userdata, flags, rc):
    print(f"[MQTT] on_connect 回调, rc={rc}")
    if rc == 0:
        print("[MQTT] 连接成功")
        client.subscribe(RECV_TOPIC)
        print(f"[MQTT] 已订阅接收主题: {RECV_TOPIC}")
    else:
        print(f"[MQTT] 连接失败, rc={rc}")

def on_message(client, userdata, msg):
    print(f"[MQTT] 收到消息: {msg.topic} -> {msg.payload.decode()}")

client.on_connect = on_connect
client.on_message = on_message
client.on_log = lambda c, u, l, b: print(f"[MQTT LOG] {b}")

# ---------- 连接服务器 ----------
try:
    rc = client.connect(MQTT_BROKER, MQTT_PORT, 60)
    print(f"[MQTT] connect() 返回码: {rc}")
except Exception as e:
    print("[MQTT] 连接异常:", e)

client.loop_start()

# ---------- 主循环：读取 DHT11 并发送 ----------
print("开始读取 DHT11 温湿度数据，按 Ctrl+C 停止")

while True:
    try:
        temperature = dhtDevice.temperature
        humidity = dhtDevice.humidity

        if temperature is not None and humidity is not None:
            print(f"[DHT11] 温度: {temperature}°C  湿度: {humidity}%")

            payload = {
                "property": [
                    {"key": "temperature", "value": temperature},
                    {"key": "humidity", "value": humidity}
                ]
            }

            result = client.publish(SEND_TOPIC, json.dumps(payload))
            print(f"[MQTT] 发送结果: rc={result.rc}")
        else:
            print("[DHT11] 读取失败，重试中...")

    except RuntimeError:
        print("[DHT11] 读取异常，重试中...")
    except Exception as e:
        print("[ERROR] 严重错误:", e)

    time.sleep(2)
